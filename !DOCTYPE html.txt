// Intentar conectar a Firebase y cargar datos
async function initFirebase() {
    if (!db) return false;
    try {
        // Intentar leer con timeout
        const snap = await Promise.race([
            db.ref('users').once('value'),
            new Promise((_, rej) => setTimeout(() => rej(new Error('timeout')), 6000))
        ]);

        App.useFirebase = true;
        App.firebaseReady = true;

        // Si no hay datos, sembrar iniciales
        if (!snap.exists() || Object.keys(snap.val() || {}).length === 0) {
            await seedFirebase();
        }

        // Cargar datos
        const [cSnap, sSnap, aSnap, uSnap] = await Promise.all([
            db.ref('courses').once('value'),
            db.ref('students').once('value'),
            db.ref('annotations').once('value'),
            db.ref('users').once('value')
        ]);

        App.courses = cSnap.val() || {};
        App.students = sSnap.val() || {};
        App.annotations = aSnap.val() || {};

        // Escuchar cambios en tiempo real
        db.ref('courses').on('value', s => { App.courses = s.val() || {}; if (App.currentUser) renderSection(); });
        db.ref('students').on('value', s => { App.students = s.val() || {}; if (App.currentUser) renderSection(); });
        db.ref('annotations').on('value', s => { App.annotations = s.val() || {}; if (App.currentUser) renderSection(); });

        updateConnStatus(true);
        return true;
    } catch (err) {
        console.warn('Firebase no disponible, usando localStorage:', err.message);
        App.useFirebase = false;
        loadFromLocalStorage();
        updateConnStatus(false);
        return false;
    }
}

async function createCourse(name, description) {
    const data = { name, description };
    if (App.useFirebase) {
        const ref = db.ref('courses').push();
        await ref.set(data);
    } else {
        App.courses[genId()] = data;
        saveToLocalStorage();
        renderSection();
    }
}

async function updateCourse(id, name, description) {
    const data = { name, description };
    if (App.useFirebase) {
        await db.ref('courses/' + id).update(data);
    } else {
        App.courses[id] = { ...App.courses[id], ...data };
        saveToLocalStorage();
        renderSection();
    }
}

async function removeCourse(id, action, newCourseId) {
    // Obtener estudiantes del curso
    const courseStudents = Object.entries(App.students).filter(([, s]) => s.courseId === id);

    if (App.useFirebase) {
        const updates = {};
        // Eliminar curso
        updates['courses/' + id] = null;

        if (action === 'reassign' && newCourseId) {
            // Reasignar estudiantes
            courseStudents.forEach(([sid]) => {
                updates['students/' + sid + '/courseId'] = newCourseId;
            });
        } else if (action === 'delete') {
            // Eliminar estudiantes y sus anotaciones
            courseStudents.forEach(([sid]) => {
                updates['students/' + sid] = null;
                // Eliminar anotaciones de esos estudiantes
                Object.entries(App.annotations).forEach(([aid, a]) => {
                    if (a.studentId === sid) updates['annotations/' + aid] = null;
                });
            });
        }

        await db.ref().update(updates);
    } else {
        delete App.courses[id];
        if (action === 'reassign' && newCourseId) {
            courseStudents.forEach(([sid]) => {
                App.students[sid].courseId = newCourseId;
            });
        } else if (action === 'delete') {
            courseStudents.forEach(([sid]) => {
                delete App.students[sid];
                // Eliminar anotaciones
                Object.entries(App.annotations).forEach(([aid, a]) => {
                    if (a.studentId === sid) delete App.annotations[aid];
                });
            });
        }
        saveToLocalStorage();
        renderSection();
    }
}

// ═══════════════ HELPERS ═══════════════
function getStudentName(id) {
    const s = App.students[id];
    return s ? `${s.name} ${s.lastname}` : 'Desconocido';
}

function getCourseName(id) {
    const c = App.courses[id];
    return c ? c.name : 'Sin curso';
}

function getStudentAnnotations(sid) {
    return Object.entries(App.annotations).filter(([, a]) => a.studentId === sid);
}

function getCourseStudents(cid) {
    return Object.entries(App.students).filter(([, s]) => s.courseId === cid);
}

function typeLabel(t) {
    const map = { positive: 'Participación +', negative: 'Participación -', task: 'Tarea pendiente', observation: 'Observación' };
    return map[t] || t;
}

function typeBadge(t) {
    const map = { positive: 'badge-positive', negative: 'badge-negative', task: 'badge-task', observation: 'badge-observation' };
    return map[t] || '';
}

function typeIcon(t) {
    const map = { positive: 'fa-thumbs-up', negative: 'fa-thumbs-down', task: 'fa-list-check', observation: 'fa-eye' };
    return map[t] || 'fa-note';
}

function formatDate(d) {
    if (!d) return '';
    const parts = d.split('-');
    if (parts.length !== 3) return d;
    return `${parts[2]}/${parts[1]}/${parts[0]}`;
}

// ═══════════════ TOAST ═══════════════
function toast(msg, type = 'info') {
    const box = document.getElementById('toastBox');
    const el = document.createElement('div');
    el.className = `toast toast-${type}`;
    const icons = { success: 'fa-check-circle', error: 'fa-times-circle', warning: 'fa-exclamation-triangle', info: 'fa-info-circle' };
    el.innerHTML = `<i class="fas ${icons[type] || icons.info}"></i> ${msg}`;
    box.appendChild(el);
    setTimeout(() => { el.classList.add('out'); setTimeout(() => el.remove(), 300); }, 3000);
}

function filterStudents(q) {
    const rows = document.querySelectorAll('#studentsTable tbody tr');
    const query = q.toLowerCase();
    rows.forEach(row => {
        const text = row.textContent.toLowerCase();
        row.style.display = text.includes(query) ? '' : 'none';
    });
}

function confirmDeleteCourse(id) {
    const c = App.courses[id];
    if (!c) return;
    const stCount = getCourseStudents(id).length;
    const otherCourses = Object.entries(App.courses).filter(([cid]) => cid !== id);
    const otherOpts = otherCourses.map(([cid, co]) => `<option value="${cid}">${co.name}</option>`).join('');

    showModal(`
        <h3 class="text-lg font-bold mb-3">Eliminar Curso</h3>
        <p class="mb-2" style="color:var(--muted)">Estás a punto de eliminar el curso <strong style="color:var(--fg)">${c.name}</strong>.</p>
        ${stCount > 0 ? `
            <p class="text-sm mb-3" style="color:var(--amber)"><i class="fas fa-exclamation-triangle mr-1"></i> Este curso tiene ${stCount} estudiante${stCount > 1 ? 's' : ''}.</p>
            <div class="space-y-3 mb-4">
                <label class="flex items-center gap-2 cursor-pointer">
                    <input type="radio" name="delAction" value="reassign" checked class="accent-[var(--accent)]">
                    <span class="text-sm">Reasignar estudiantes a otro curso</span>
                </label>
                <div id="reassignDiv" class="pl-6">
                    <select id="reassignCourse" class="ipt"><option value="">Seleccionar curso</option>${otherOpts}</select>
                </div>
                <label class="flex items-center gap-2 cursor-pointer">
                    <input type="radio" name="delAction" value="delete" class="accent-[var(--accent)]">
                    <span class="text-sm" style="color:var(--danger)">Eliminar estudiantes y sus anotaciones</span>
                </label>
            </div>
        ` : ''}
        <div class="flex justify-end gap-3 mt-5">
            <button class="btn btn-ghost" onclick="hideModal()">Cancelar</button>
            <button class="btn btn-danger" onclick="doDeleteCourse('${id}')">Eliminar Curso</button>
        </div>
    `);

    // Toggle reassign select visibility
    setTimeout(() => {
        document.querySelectorAll('input[name="delAction"]').forEach(r => {
            r.addEventListener('change', () => {
                const div = document.getElementById('reassignDiv');
                if (div) div.style.display = r.value === 'reassign' && r.checked ? '' : 'none';
            });
        });
    }, 50);
}

async function doDeleteCourse(id) {
    const stCount = getCourseStudents(id).length;
    let action = 'delete';
    let newCourseId = '';

    if (stCount > 0) {
        const checked = document.querySelector('input[name="delAction"]:checked');
        action = checked ? checked.value : 'delete';
        if (action === 'reassign') {
            newCourseId = document.getElementById('reassignCourse')?.value || '';
            if (!newCourseId) {
                toast('Selecciona un curso para reasignar', 'error');
                return;
            }
        }
    }

    await removeCourse(id, action, newCourseId);
    hideModal();
    toast('Curso eliminado correctamente', 'success');
}

                const task = anns.filter(([,a]) => a.type === 'task').length;
                const obs = anns.filter(([,a]) => a.type === 'observation').length;
                html += `<tr><td>${s.name} ${s.lastname}</td><td>${s.email}</td><td style="color:#059669">${pos}</td><td style="color:#dc2626">${neg}</td><td style="color:#d97706">${task}</td><td style="color:#2563eb">${obs}</td></tr>`;
            });
            html += `</tbody></table>`;
        }
    });

    area.innerHTML = html;

    try {
        // Esperar un poco para que el contenido se renderice
        await new Promise(r => setTimeout(r, 200));

        const canvas = await html2canvas(area, { scale: 2, useCORS: true, backgroundColor: '#ffffff' });
        const { jsPDF } = window.jspdf;
        const pdf = new jsPDF('p', 'mm', 'a4');

        const imgWidth = 210;
        const pageHeight = 297;
        const imgHeight = (canvas.height * imgWidth) / canvas.width;
        let heightLeft = imgHeight;
        let position = 0;

        pdf.addImage(canvas.toDataURL('image/png'), 'PNG', 0, position, imgWidth, imgHeight);
        heightLeft -= pageHeight;

        while (heightLeft > 0) {
            position = heightLeft - imgHeight;
            pdf.addPage();
            pdf.addImage(canvas.toDataURL('image/png'), 'PNG', 0, position, imgWidth, imgHeight);
            heightLeft -= pageHeight;
        }

        pdf.save('EduTrack_Reporte.pdf');
        area.innerHTML = '';
        toast('PDF generado correctamente', 'success');
    } catch (err) {
        console.error('Error al generar PDF:', err);
        area.innerHTML = '';
        toast('Error al generar PDF', 'error');
    }
}

// ═══════════════ EXPORTAR / IMPORTAR RESPALDO ═══════════════
function exportBackup() {
    const data = {
        exportDate: new Date().toISOString(),
        courses: App.courses,
        students: App.students,
        annotations: App.annotations
    };
    const blob = new Blob([JSON.stringify(data, null, 2)], { type: 'application/json' });
    const url = URL.createObjectURL(blob);
    const a = document.createElement('a');
    a.href = url;
    a.download = `edutrack_backup_${new Date().toISOString().slice(0,10)}.json`;
    a.click();
    URL.revokeObjectURL(url);
    toast('Respaldo exportado correctamente', 'success');
}

function importBackup(event) {
    const file = event.target.files[0];
    if (!file) return;

    const reader = new FileReader();
    reader.onload = async (e) => {
        try {
            const data = JSON.parse(e.target.result);
            if (!data.courses || !data.students || !data.annotations) {
                toast('Archivo de respaldo inválido', 'error');
                return;
            }

            showModal(`
                <h3 class="text-lg font-bold mb-3">Importar Respaldo</h3>
                <p class="mb-2" style="color:var(--muted)">Se reemplazarán todos los datos actuales con los del archivo.</p>
                <p class="text-sm mb-4" style="color:var(--amber)"><i class="fas fa-exclamation-triangle mr-1"></i> Esta acción no se puede deshacer.</p>
                <div class="flex justify-end gap-3">
                    <button class="btn btn-ghost" onclick="hideModal()">Cancelar</button>
                    <button class="btn btn-primary" onclick="doImportBackup()">Importar</button>
                </div>
            `);

            window._pendingImport = data;
        } catch (err) {
            toast('Error al leer el archivo', 'error');
        }
    };
    reader.readAsText(file);
    event.target.value = ''; // Resetear input
}

async function doImportBackup() {
    const data = window._pendingImport;
    if (!data) return;

    if (App.useFirebase) {
        await db.ref('courses').set(data.courses);
        await db.ref('students').set(data.students);
        await db.ref('annotations').set(data.annotations);
    } else {
        App.courses = data.courses;
        App.students = data.students;
        App.annotations = data.annotations;
        saveToLocalStorage();
    }

    delete window._pendingImport;
    hideModal();
    renderSection();
    toast('Respaldo importado correctamente', 'success');
}
